<TabItem value="yaml" label="Erzeugtes YAML">

```yaml
apiVersion: image.openshift.io/v1
kind: ImageStream
metadata:
  name: api
  namespace: api-dev
spec:
  lookupPolicy:
    local: true
---
apiVersion: build.openshift.io/v1
kind: BuildConfig
metadata:
  name: api
  namespace: api-dev
spec:
  runPolicy: Serial
  source:
    type: Git
    git:
      uri: https://github.com/example/api.git
      ref: develop
  strategy:
    type: Source
    sourceStrategy:
      from:
        kind: ImageStreamTag
        name: nodejs:20-ubi9
        namespace: openshift
  output:
    to:
      kind: ImageStreamTag
      name: api:latest
  triggers:
    - type: ConfigChange
    - type: GitHub
      github:
        secretReference:
          name: api-webhook
---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: api
  namespace: api-dev
  annotations:
    image.openshift.io/triggers: '[{"from":{"kind":"ImageStreamTag","name":"api:latest"},"fieldPath":"spec.template.spec.containers[?(@.name==\"api\")].image"}]'
spec:
  replicas: 1
  selector:
    matchLabels:
      app: api
  template:
    metadata:
      labels:
        app: api
    spec:
      containers:
        - name: api
          image: api:latest
          ports:
            - name: http
              containerPort: 8080
---
apiVersion: v1
kind: Service
metadata:
  name: api
  namespace: api-dev
spec:
  selector:
    app: api
  ports:
    - name: http
      port: 8080
      targetPort: http
  type: ClusterIP
---
apiVersion: route.openshift.io/v1
kind: Route
metadata:
  name: api
  namespace: api-dev
spec:
  host: api.apps.dev.example.com
  to:
    kind: Service
    name: api
  port:
    targetPort: http
  tls:
    termination: edge
    insecureEdgeTerminationPolicy: Redirect
```
